% Lista de siglas em ordem alfabética.

\begin{siglas}
    \item[\textit{ARM}]      \textit{Advanced RISC Machine} (Máquina RISC Avançada)
    \item[\textit{CIFS}]     \textit{Common Internet File System} (Sistema de Arquivos Comum da Internet)
    \item[\textit{CPU}]      \textit{Central Processing Unit} (Unidade Central de Processamento)
    \item[\textit{FTP}]      \textit{File Transfer Protocol} (Protocolo de Transferência de Arquivos)
    \item[\textit{HTTP}]     \textit{Hypertext Transfer Protocol} (Protocolo de Transferência de Hipertexto)
    \item[\textit{HTTPS}]    \textit{Hypertext Transfer Protocol Secure} (Protocolo Seguro de Transferência de Hipertexto)
    \item[\textit{IDC}]      \textit{International Data Corporation} (Corporação Internacional de Dados)
    \item[\textit{LAN}]      \textit{Local Area Network} (Rede de Área Local)
    \item[\textit{NAS}]      \textit{Network Attached Storage} (Armazenamento Conectado à Rede)
    \item[\textit{NFS}]      \textit{Network File System} (Sistema de Arquivos de Rede)
    \item[\textit{OMV}]      \textit{OpenMediaVault} (\textit{software} livre de administração de \textit{NAS}; sem tradução)
    \item[\textit{RAM}]      \textit{Random Access Memory} (Memória de Acesso Aleatório)
    \item[\textit{SATA}]     \textit{Serial Advanced Technology Attachment} (Tecnologia Avançada de Anexação Serial)
    \item[\textit{SBC}]      \textit{Single Board Computer} (Computador de Placa Única)
    \item[\textit{SMB}]      \textit{Server Message Block} (Bloco de Mensagens do Servidor)
    \item[\textit{SSD}]      \textit{Solid State Drive} (Unidade de Estado Sólido)
    \item[\textit{SSH}]      \textit{Secure Shell} (Intérprete Seguro de Comandos)
    \item[\textit{TCP}]      \textit{Transmission Control Protocol} (Protocolo de Controle de Transmissão)
    \item[\textit{USB}]      \textit{Universal Serial Bus} (Barramento Serial Universal)
    \item[\textit{VPN}]      \textit{Virtual Private Network} (Rede Privada Virtual)
    \item[\textit{WebDAV}]   \textit{Web-based Distributed Authoring and Versioning} (Autoria e Versionamento Distribuídos na Web)
\end{siglas}
